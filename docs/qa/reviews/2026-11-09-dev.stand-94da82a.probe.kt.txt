@RunWith(AndroidJUnit4::class)
class QaCornerProbeTest {
    @get:Rule val compose = createComposeRule()
    @Test fun controlUnderTheCornerStillGetsTaps() {
        var taps = 0
        val owner = object : LifecycleOwner { override val lifecycle: Lifecycle get() = lc }
        lc = LifecycleRegistry.createUnsafe(owner); lc.currentState = Lifecycle.State.RESUMED
        compose.setContent {
            StandHostContent(StandSettings.of(ApplicationProvider.getApplicationContext()), object : ScreenLock {
                override fun isLocked() = false; override fun lock() = true; override fun unlock() {}
            }, lc, { 0L }, false, {}) {
                Box(Modifier.fillMaxSize()) {
                    Box(Modifier.align(Alignment.BottomEnd).size(48.dp).testTag("under").clickable { taps++ })
                }
            }
        }
        compose.onNodeWithTag("under").performTouchInput { click(center) }
        assertEquals("a button in the bottom-end corner did not get its tap", 1, taps)
    }
    private lateinit var lc: LifecycleRegistry
}
